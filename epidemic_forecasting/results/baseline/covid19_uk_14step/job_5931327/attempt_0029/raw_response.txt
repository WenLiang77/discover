```python
import numpy as np
from statsmodels.tsa.statespace.sarimax import SARIMAX

def covid_forecast(train_values, horizon, **kwargs):
    # Initialize an empty array to store the forecasted values
    forecast = np.zeros((horizon, train_values.shape[1]))
    
    # Loop through each region
    for i in range(train_values.shape[1]):
        # Fit a SARIMAX model to the historical data for the current region
        try:
            model = SARIMAX(train_values[:, i], order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            model_fit = model.fit(disp=False)
        except Exception as e:
            # If fitting fails, use a simple exponential smoothing approach as a fallback
            model_fit = None
        
        if model_fit is not None:
            # Generate forecasts for the specified horizon
            forecast[:, i] = model_fit.forecast(steps=horizon)
        
        # Ensure all forecasted values are finite and non-negative
        forecast[:, i] = np.clip(forecast[:, i], 0, None)
    
    return forecast
```
