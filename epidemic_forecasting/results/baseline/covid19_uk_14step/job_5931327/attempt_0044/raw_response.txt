import numpy as np
from statsmodels.tsa.statespace.sarimax import SARIMAX

def covid_forecast(train_values, horizon, **kwargs):
    # Initialize forecasts with zeros
    forecasts = np.zeros((horizon, train_values.shape[1]))
    
    # Loop over each region to fit a SARIMA model and make forecasts
    for i in range(train_values.shape[1]):
        try:
            # Fit a SARIMAX model on the historical data for the current region
            model = SARIMAX(train_values[:, i], order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            fitted_model = model.fit(disp=False)
            
            # Make forecasts for the specified horizon
            forecast = fitted_model.get_forecast(steps=horizon).predicted_mean
            
            # Ensure forecasts are non-negative
            forecast = np.maximum(forecast, 0)
            
            # Store the forecasts for the current region
            forecasts[:, i] = forecast
        except Exception as e:
            # If fitting fails, fill with zeros (or could handle differently)
            forecasts[:, i] = np.zeros(horizon)
    
    return forecasts
