```python
import numpy as np
from statsmodels.tsa.statespace.sarimax import SARIMAX
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures

def dengue_forecast(train_values, horizon, **kwargs):
    forecast = np.zeros((horizon, train_values.shape[1]))
    
    for i in range(train_values.shape[1]):
        try:
            if np.any(train_values[:, i] <= 0):
                train_values[:, i] = np.where(train_values[:, i] <= 0, np.nan, train_values[:, i])
            
            # Fit a SARIMAX model
            model = SARIMAX(train_values[:, i], order=(1, 1, 0), seasonal_order=(1, 1, 0, 4), enforce_stationarity=False, enforce_invertibility=False)
            results = model.fit(disp=False)
            forecast[:, i] = results.get_forecast(steps=horizon).predicted_mean
            
            # Ensure forecast is non-negative
            forecast[:, i] = np.maximum(forecast[:, i], 0)
        
        except Exception as e:
            try:
                # If SARIMAX fails, fit a polynomial regression model
                X_train = np.arange(len(train_values)).reshape(-1, 1)
                y_train = train_values[:, i].reshape(-1, 1)
                
                # Add polynomial features
                poly_features = PolynomialFeatures(degree=2)
                X_train_poly = poly_features.fit_transform(X_train)
                
                # Fit linear regression model
                lin_reg = LinearRegression()
                lin_reg.fit(X_train_poly, y_train)
                
                # Create forecast indices
                X_forecast = np.arange(len(train_values), len(train_values) + horizon).reshape(-1, 1)
                X_forecast_poly = poly_features.transform(X_forecast)
                
                # Predict forecast values
                forecast[:, i] = lin_reg.predict(X_forecast_poly)
                
                # Ensure forecast is non-negative
                forecast[:, i] = np.maximum(forecast[:, i], 0)
            
            except Exception as e:
                # Fallback: interpolate missing values with regional means
                regional_means = np.nanmean(train_values[:, train_values[:, i] > 0], axis=1)
                forecast[:, i] = np.interp(range(horizon), [0, len(regional_means) - 1], regional_means)
    
    return forecast
```

This implementation tries to fit a SARIMAX model to each region's dengue case count data. If the model fails (e.g., due to insufficient data or model instability), it falls back to fitting a polynomial regression model on the historical data. If both methods fail, it interpolates the missing values with the regional means. This approach ensures that the forecast is non-negative and handles various edge cases gracefully.
