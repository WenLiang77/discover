from statsmodels.tsa.statespace.sarimax import SARIMAX

def dengue_forecast(train_values, horizon, **kwargs):

            if np.any(train_values[:, i] <= 0):

            forecast[:, i] = results.get_forecast(steps=horizon).predicted_mean

                forecast[:, i] = scaler.inverse_transform(forecast_scaled)
            except Exception as e:
                # If linear regression also fails, use a simple interpolation based on regional means
                if np.all(np.isnan(train_values[:, i])):
                    regional_means = np.nanmean(train_values[:, train_values[:, i] > 0], axis=1)
                    forecast[:, i] = np.interp(range(horizon), [0, len(regional_means) - 1], regional_means)

    # Ensure the forecast is non-negative
    forecast = np.maximum(forecast, 0)

This implementation includes a SARIMA model fit for each region, which captures temporal dynamics and seasonality. If the SARIMA model fitting fails, it falls back to a linear regression model. If both fail, it uses a simple interpolation based on regional means. This approach ensures that the forecast is non-negative and has the required shape.
