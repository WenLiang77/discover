```python
import numpy as np
from statsmodels.tsa.statespace.sarimax import SARIMAX
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.arima.model import ARIMA
from sklearn.linear_model import LinearRegression
from sklearn.preprocessing import PolynomialFeatures
from math import inf
from typing import Optional

def dengue_forecast(
    train_values: np.ndarray,
    horizon: int,
    random_state: Optional[int] = None,
) -> np.ndarray:
    n_regions = train_values.shape[1]
    forecast = np.empty((horizon, n_regions))

    for i in range(n_regions):
        region_data = train_values[:, i]

        # Check if the series is non-zero
        if np.any(region_data > 0):
            try:
                # Fit SARIMAX model
                model_sarimax = SARIMAX(region_data, order=(1, 1, 0), seasonal_order=(1, 1, 0, 4))
                results_sarimax = model_sarimax.fit(disp=False)
                forecast[i, :] = results_sarimax.forecast(steps=horizon)

            except Exception as e:
                # If SARIMAX fails, fall back to Exponential Smoothing
                try:
                    model_es = ExponentialSmoothing(region_data, trend='add', seasonal='add', seasonal_periods=4)
                    results_es = model_es.fit()
                    forecast[i, :] = results_es.forecast(steps=horizon)

                except Exception as e:
                    # If Exponential Smoothing fails, fall back to ARIMA
                    try:
                        model_arima = ARIMA(region_data, order=(1, 1, 0))
                        results_arima = model_arima.fit(disp=False)
                        forecast[i, :] = results_arima.forecast(steps=horizon)

                    except Exception as e:
                        # If ARIMA fails, fall back to simple linear regression
                        if len(region_data) > 2:
                            X = np.arange(len(region_data)).reshape(-1, 1)
                            y = region_data
                            poly_features = PolynomialFeatures(degree=2)
                            X_poly = poly_features.fit_transform(X)
                            model_lr = LinearRegression()
                            model_lr.fit(X_poly, y)
                            X_pred = np.array(range(len(region_data), len(region_data) + horizon)).reshape(-1, 1)
                            X_pred_poly = poly_features.transform(X_pred)
                            forecast[i, :] = model_lr.predict(X_pred_poly)
                        else:
                            forecast[i, :] = region_data[-1]

        else:
            # If the series is constant or zero, predict the last observed value
            forecast[i, :] = region_data[-1]

    # Ensure non-negativity
    forecast = np.maximum(forecast, 0)

    return forecast
```

This implementation includes several steps to improve upon the initial candidate:

1. **SARIMAX Model**: Fits a Seasonal AutoRegressive Integrated Moving Average with eXogenous factors (SARIMAX) model, which accounts for both seasonal and trend effects.

2. **Exponential Smoothing**: If SARIMAX fails, falls back to Exponential Smoothing, which is useful for capturing trends and seasonal patterns without requiring a large amount of data.

3. **ARIMA Model**: If Exponential Smoothing fails, fits an ARIMA model, which is suitable for univariate time series analysis.

4. **Linear Regression**: As a last resort, uses polynomial regression to fit a quadratic curve to the data. This approach is less sophisticated but ensures that the forecast is at least non-negative.

5. **Constant Series Handling**: If the series is constant or zero, predicts the last observed value.

6. **Non-Negative Constraints**: Ensures that all forecasts are non-negative using `np.maximum(forecast, 0)`.

This approach provides a robust and versatile forecasting method that can handle various types of time series data, including those with limited historical observations or non-stationarity.
