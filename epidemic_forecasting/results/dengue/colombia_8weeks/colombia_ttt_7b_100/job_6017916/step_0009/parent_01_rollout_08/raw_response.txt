import numpy as np
from statsmodels.tsa.statespace.sarimax import SARIMAX

def dengue_forecast(train_values, horizon, **kwargs):
    num_regions = train_values.shape[1]
    forecasted_values = np.zeros((horizon, num_regions), dtype=float)
    
    for region in range(num_regions):
        region_data = train_values[:, region]
        
        # Check if differencing is needed
        diff_order = 0
        if np.diff(region_data).var() > 0.01 * region_data.var():
            diff_order = 1

        try:
            model = SARIMAX(region_data, order=(1, diff_order, 1), seasonal_order=(1, 0, 1, 52)).fit(disp=False)

### Explanation:
1. **Initialization**: We initialize an empty array `forecasted_values` to store the predicted values.
2. **Loop through each region**: For each region, we extract the historical data.

4. **Fit SARIMAX model**: We attempt to fit a SARIMAX model. If fitting fails (e.g., due to insufficient data or other issues), we fall back to an Exponential Smoothing model.
