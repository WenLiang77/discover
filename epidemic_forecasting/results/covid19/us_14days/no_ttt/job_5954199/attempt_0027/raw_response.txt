```python
import numpy as np
from statsmodels.tsa.statespace.sarimax import SARIMAX
from sklearn.metrics import mean_absolute_error, mean_squared_error

def covid_forecast(train_values, horizon, **kwargs):
    # Ensure train_values is a NumPy array
    train_values = np.array(train_values)
    
    # Get the number of time steps and regions
    T, N = train_values.shape
    
    # Initialize the forecasted values array
    forecasts = np.zeros((horizon, N))
    
    # Loop through each region to fit a SARIMA model and make predictions
    for region in range(N):
        try:
            # Fit a SARIMA model to the historical data for the current region
            model = SARIMAX(train_values[:, region], order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            fitted_model = model.fit(disp=False)
            
            # Make predictions for the specified horizon
            forecast = fitted_model.get_forecast(steps=horizon).predicted_mean
            
            # Ensure the forecasted values are non-negative
            forecast[forecast < 0] = 0
            
            # Store the forecasted values
            forecasts[:, region] = forecast
        except Exception as e:
            # If fitting fails, fill the forecast with zeros
            forecasts[:, region] = 0
            print(f"Failed to fit model for region {region}: {e}")
    
    return forecasts
```

This implementation fits a SARIMA model to the historical data for each region and makes predictions for the specified horizon. It ensures that all predicted values are non-negative and handles fitting failures by filling the forecast with zeros.
