```python
import numpy as np
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.statespace.sarimax import SARIMAX
from statsmodels.tsa.seasonal import seasonal_decompose
from sklearn.preprocessing import MinMaxScaler

def covid_forecast(train_values, horizon, **kwargs):
    forecasted_values = np.zeros((horizon, train_values.shape[1]))
    
    # Initialize scaler
    scaler = MinMaxScaler(feature_range=(0, 1))
    
    for region in range(train_values.shape[1]):
        region_data = train_values[:, region]
        
        # Scale the data
        scaled_data = scaler.fit_transform(region_data.reshape(-1, 1))
        
        try:
            # Decompose the data into seasonal and trend components
            decomposition = seasonal_decompose(scaled_data.flatten(), model='additive', period=7)
            
            # Fit SARIMAX model on the decomposed data
            sarimax_model = SARIMAX(decomposition.resid, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            sarimax_fitted = sarimax_model.fit(disp=False)
            
            # Get forecast from SARIMAX model
            sarimax_forecast = sarimax_fitted.get_forecast(steps=horizon)
            sarimax_forecast_values = sarimax_forecast.predicted_mean
            
            # Add back the trend and seasonal components
            trend_forecast = decomposition.trend[-1] + (decomposition.trend[-1] - decomposition.trend[-2]) * np.arange(1, horizon + 1)
            seasonal_forecast = decomposition.seasonal[-7:] + (decomposition.seasonal[-7:] - decomposition.seasonal[-8:-1]) * np.arange(1, horizon + 1)
            
            forecast = sarimax_forecast_values * decomposition.seasonal[-1] + trend_forecast
            
            # Inverse transform the forecast
            forecasted_values[:, region] = scaler.inverse_transform(forecast.reshape(-1, 1))[:, 0].clip(min=0)
        
        except Exception as _:
            try:
                # If SARIMAX fails, fall back to Exponential Smoothing
                es_model = ExponentialSmoothing(scaled_data.flatten(), trend='add', seasonal=None)
                es_fitted = es_model.fit()
                
                # Get forecast from Exponential Smoothing model
                es_forecast = es_fitted.forecast(steps=horizon)
                
                # Inverse transform the forecast
                forecasted_values[:, region] = scaler.inverse_transform(es_forecast.reshape(-1, 1))[:, 0].clip(min=0)
            except Exception as _:
                # If both models fail, fill with zeros
                forecasted_values[:, region] = 0
    
    return forecasted_values
```

### Explanation:
1. **Scaling**: The data is scaled using `MinMaxScaler` to handle potential outliers and to ensure numerical stability.
2. **Decomposition**: The data is decomposed into seasonal and trend components using `seasonal_decompose`.
3. **SARIMAX Model**: A SARIMAX model is fit on the residuals of the decomposition. This helps capture both trend and seasonality.
4. **Forecasting**: Forecasts are obtained from the SARIMAX model, and then adjusted by adding back the trend and seasonal components.
5. **Inverse Transformation**: The forecasts are inverse transformed back to the original scale and clipped to non-negative values.
6. **Fallback Mechanism**: If the SARIMAX model fails, an Exponential Smoothing model is used as a fallback. If both models fail, the forecast is filled with zeros.

This approach ensures that the model can handle various characteristics of the data, including non-stationarity and seasonality, while being robust to fitting failures.
