```python
import numpy as np
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.tsa.statespace.sarimax import SARIMAX
from statsmodels.tsa.seasonal import seasonal_decompose
from sklearn.preprocessing import PowerTransformer
from sklearn.impute import SimpleImputer
from scipy.stats.mstats import winsorize

def covid_forecast(train_values, horizon, **kwargs):
    forecasted_values = np.zeros((horizon, train_values.shape[1]))
    
    for region in range(train_values.shape[1]):
        region_data = train_values[:, region].copy()
        
        # Handle missing values
        imputer = SimpleImputer(strategy='median')
        region_data = imputer.fit_transform(region_data.reshape(-1, 1)).flatten()
        
        # Winsorize extreme values
        region_data = winsorize(region_data, limits=[0.05, 0.05])
        
        # Decompose the time series to identify seasonal components
        decomposition = seasonal_decompose(region_data, model='additive', period=7)
        trend = decomposition.trend
        seasonal = decomposition.seasonal
        
        # Combine trend and seasonal components for better prediction
        combined_data = (trend + seasonal).clip(0)
        
        try:
            # Fit SARIMAX model
            model = SARIMAX(combined_data, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            fitted_model = model.fit(disp=False)
            forecast = fitted_model.get_forecast(steps=horizon).predicted_mean
            
            # Apply inverse transformation
            forecasted_values[:, region] = np.maximum(forecast, 0)
        except Exception as _:
            pass
        
        if np.allclose(forecasted_values[:, region], 0):
            try:
                # Fit ARIMA model if SARIMAX fails
                model = ARIMA(combined_data, order=(1, 1, 1))
                fitted_model = model.fit(disp=False)
                forecast = fitted_model.forecast(steps=horizon)
                
                # Apply inverse transformation
                forecasted_values[:, region] = np.maximum(forecast, 0)
            except Exception as _:
                pass
        
        if np.allclose(forecasted_values[:, region], 0):
            try:
                # Fit Exponential Smoothing if ARIMA also fails
                model = ExponentialSmoothing(combined_data, trend='add', seasonal=None)
                fitted_model = model.fit()
                forecast = fitted_model.forecast(steps=horizon)
                
                # Apply inverse transformation
                forecasted_values[:, region] = np.maximum(forecast, 0)
            except Exception as _:
                pass
    
    return forecasted_values
```
