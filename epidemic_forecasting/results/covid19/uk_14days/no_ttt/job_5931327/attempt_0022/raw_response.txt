```python
import numpy as np
from statsmodels.tsa.holtwinters import ExponentialSmoothing

def covid_forecast(train_values, horizon, **kwargs):
    # Initialize the forecasted values array
    forecasts = np.zeros((horizon, train_values.shape[1]))
    
    for region in range(train_values.shape[1]):
        try:
            # Fit an Exponential Smoothing model to each region's time series
            model = ExponentialSmoothing(train_values[:, region], trend='add', seasonal=None)
            fitted_model = model.fit()
            
            # Predict the next 'horizon' time steps
            forecast_region = fitted_model.forecast(steps=horizon)
            
            # Ensure the forecast values are non-negative
            forecast_region = np.maximum(forecast_region, 0)
            
            # Assign the forecast to the corresponding region
            forecasts[:, region] = forecast_region
        except Exception as e:
            # If fitting fails, fill the forecast with zeros
            forecasts[:, region] = 0
    
    return forecasts
```
