```python
import numpy as np
from statsmodels.tsa.statespace.sarimax import SARIMAX

def covid_forecast(train_values, horizon, **kwargs):
    # Initialize predictions array with zeros
    predictions = np.zeros((horizon, train_values.shape[1]))
    
    # Iterate over each region
    for region in range(train_values.shape[1]):
        # Fit SARIMAX model on the historical data for the current region
        try:
            model = SARIMAX(train_values[:, region], order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            fitted_model = model.fit(disp=False)
            
            # Predict future values
            forecast = fitted_model.forecast(steps=horizon)
            predictions[:, region] = forecast.clip(0)  # Clip negative predictions to zero
        except Exception as e:
            # If fitting fails, fill the region's predictions with zeros
            print(f"Model fitting failed for region {region}: {e}")
            predictions[:, region] = 0
    
    return predictions
```
