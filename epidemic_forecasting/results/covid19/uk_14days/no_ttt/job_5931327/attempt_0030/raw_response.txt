import numpy as np
from statsmodels.tsa.statespace.sarimax import SARIMAX

def covid_forecast(train_values, horizon, **kwargs):
    # Ensure train_values is a numpy array
    train_values = np.array(train_values)
    
    # Get the number of regions (N) and time steps (T)
    N, T = train_values.shape
    
    # Initialize the forecasted values array
    forecasted_values = np.zeros((horizon, N))
    
    # Loop over each region to fit a SARIMA model and make predictions
    for region in range(N):
        try:
            # Fit a SARIMA model to the historical data for the current region
            model = SARIMAX(train_values[region], order=(5, 1, 0), seasonal_order=(1, 1, 1, 7))
            results = model.fit(disp=False)
            
            # Make predictions for the future time steps
            forecast = results.get_forecast(steps=horizon).predicted_mean
            
            # Ensure the forecasted values are non-negative
            forecasted_values[:, region] = np.maximum(forecast, 0)
        except Exception as e:
            # If fitting fails, fill the forecast with zeros
            forecasted_values[:, region] = 0
    
    return forecasted_values
