```python
import numpy as np
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.statespace.sarimax import SARIMAX
from statsmodels.tsa.seasonal import seasonal_decompose
from statsmodels.graphics.tsaplots import plot_acf, plot_pacf
import matplotlib.pyplot as plt

def decompose_and_fit(data, order, seasonal_order):
    decomposition = seasonal_decompose(data, model='additive')
    trend = decomposition.trend
    seasonal = decomposition.seasonal
    residual = decomposition.resid
    
    try:
        model = SARIMAX(data, order=order, seasonal_order=seasonal_order)
        fitted_model = model.fit(disp=False)
        return fitted_model
    except Exception as _:
        return None

def covid_forecast(train_values, horizon, **kwargs):
    forecasted_values = np.zeros((horizon, train_values.shape[1]))
    
    for region in range(train_values.shape[1]):
        region_data = train_values[:, region]
        
        # Try different combinations of ARIMA and SARIMA models
        for order in [(1, 1, 1), (2, 1, 1), (1, 2, 1)]:
            for seasonal_order in [(1, 1, 1, 7), (1, 1, 1, 14), (1, 1, 1, 30)]:
                model = decompose_and_fit(region_data, order, seasonal_order)
                if model is not None:
                    forecast = model.get_forecast(steps=horizon).predicted_mean
                    forecasted_values[:, region] = np.maximum(forecast, 0)
                    break
            else:
                continue
            break
        
        # If no model fits, use Exponential Smoothing
        if np.all(np.isnan(forecasted_values[:, region])):
            try:
                model = ExponentialSmoothing(region_data, trend='add', seasonal=None)
                fitted_model = model.fit()
                forecast = fitted_model.forecast(steps=horizon)
                forecasted_values[:, region] = np.maximum(forecast, 0)
            except Exception as _:
                pass
    
    return forecasted_values
```

This implementation attempts to fit multiple SARIMA models with different orders and seasonal orders to each region's data. If none of these models fit successfully, it falls back to using Exponential Smoothing. This approach aims to handle various types of patterns in the data, including trends and seasonality.
