```python
import numpy as np
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.statespace.sarimax import SARIMAX
from sklearn.preprocessing import StandardScaler

def covid_forecast(train_values, horizon, **kwargs):
    forecasted_values = np.zeros((horizon, train_values.shape[1]))
    scaler = StandardScaler(with_mean=False)
    
    for region in range(train_values.shape[1]):
        region_data = train_values[:, region]
        
        # Add a small constant to avoid log(0) or division by zero issues
        region_data += 1e-06
        
        # Scale the data
        region_data_scaled = scaler.fit_transform(region_data.reshape(-1, 1)).flatten()
        
        try:
            # Fit a SARIMAX model
            model = SARIMAX(region_data_scaled, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            fitted_model = model.fit(disp=False)
            forecast_scaled = fitted_model.get_forecast(steps=horizon).predicted_mean
        except Exception as _:
            try:
                # Fit an exponential smoothing model
                model = ExponentialSmoothing(region_data_scaled, trend='add', seasonal=None)
                fitted_model = model.fit()
                forecast_scaled = fitted_model.forecast(steps=horizon)
            except Exception as _:
                # If both models fail, use a simple mean-based forecast
                forecast_scaled = np.full(horizon, np.mean(region_data_scaled))
        
        # Transform back to original scale and clip negative values to zero
        forecast_original_scale = scaler.inverse_transform(forecast_scaled.reshape(-1, 1)).flatten()
        forecasted_values[:, region] = np.maximum(forecast_original_scale, 0)
    
    return forecasted_values
```

This implementation attempts to fit a SARIMAX model to each region's data, but falls back to exponential smoothing if the SARIMAX model fails. If neither model works, it uses a simple mean-based forecast. The predictions are then transformed back to the original scale and any negative values are clipped to zero. This approach aims to handle various types of time series characteristics while ensuring robustness and safety.
