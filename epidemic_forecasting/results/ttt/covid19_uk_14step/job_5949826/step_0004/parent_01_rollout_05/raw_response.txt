```python
import numpy as np
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.tsa.statespace.sarimax import SARIMAX
from statsmodels.tsa.seasonal import seasonal_decompose
from sklearn.preprocessing import PowerTransformer
from sklearn.impute import SimpleImputer
from scipy.stats.mstats import gmean
from typing import Optional

def covid_forecast(train_values: np.ndarray, horizon: int, **kwargs) -> np.ndarray:
    forecasted_values = np.zeros((horizon, train_values.shape[1]))
    
    imputer = SimpleImputer(strategy='mean')
    pt = PowerTransformer(method='yeo-johnson')
    
    for region in range(train_values.shape[1]):
        region_data = train_values[:, region].copy()
        
        # Impute missing values
        region_data = imputer.fit_transform(region_data.reshape(-1, 1)).flatten()
        
        # Transform data
        transformed_data = pt.fit_transform(region_data.reshape(-1, 1)).flatten()
        
        try:
            # Seasonal decomposition
            decomposition = seasonal_decompose(transformed_data, model='multiplicative', period=7)
            trend = decomposition.trend
            seasonal = decomposition.seasonal
            
            # SARIMAX model for trend and seasonality
            sarimax_model = SARIMAX(trend, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            sarimax_fitted_model = sarimax_model.fit(disp=False)
            
            # Forecast trend
            trend_forecast = sarimax_fitted_model.get_forecast(steps=horizon).predicted_mean
            
            # ARIMA model for residuals
            residuals = transformed_data - (trend + seasonal)
            arima_model = ARIMA(residuals, order=(1, 1, 1))
            arima_fitted_model = arima_model.fit(disp=False)
            
            # Forecast residuals
            residual_forecast = arima_fitted_model.get_forecast(steps=horizon).predicted_mean
            
            # Combine forecasts
            combined_forecast = trend_forecast + seasonal + residual_forecast
            
            # Inverse transform forecast
            inverse_transformed_forecast = pt.inverse_transform(combined_forecast.reshape(-1, 1)).flatten()
            
            # Clip negative values to zero
            forecasted_values[:, region] = np.maximum(inverse_transformed_forecast, 0)
        
        except Exception as e:
            try:
                # Use exponential smoothing if decomposition fails
                es_model = ExponentialSmoothing(region_data, trend='add', seasonal=None)
                es_fitted_model = es_model.fit()
                
                # Forecast
                forecast = es_fitted_model.forecast(steps=horizon)
                
                # Clip negative values to zero
                forecasted_values[:, region] = np.maximum(forecast, 0)
            except Exception as e:
                # Use simple moving average if everything else fails
                window_size = max(1, len(region_data) // 10)
                moving_average = np.convolve(region_data, np.ones(window_size)/window_size, mode='valid')
                padded_ma = np.pad(moving_average, (0, horizon - len(moving_average)), mode='constant')
                forecasted_values[:, region] = np.maximum(padded_ma, 0)
    
    return forecasted_values
```
