```python
import numpy as np
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.tsa.statespace.sarimax import SARIMAX
from statsmodels.tsa.seasonal import seasonal_decompose
from sklearn.preprocessing import PowerTransformer
from sklearn.impute import SimpleImputer

def covid_forecast(train_values, horizon, **kwargs):
    forecasted_values = np.zeros((horizon, train_values.shape[1]))
    imputer = SimpleImputer(strategy='median')
    for region in range(train_values.shape[1]):
        region_data = train_values[:, region]
        # Handle missing values
        region_data = imputer.fit_transform(region_data.reshape(-1, 1)).flatten()
        
        # Apply power transformation
        pt = PowerTransformer(method='yeo-johnson')
        transformed_data = pt.fit_transform(region_data.reshape(-1, 1)).flatten()
        
        # Decompose data
        try:
            decomposition = seasonal_decompose(transformed_data, model='additive', period=7)
            trend = decomposition.trend
            seasonal = decomposition.seasonal
            
            # Fit SARIMAX model
            sarimax_model = SARIMAX(trend, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            sarimax_fitted_model = sarimax_model.fit(disp=False)
            
            # Get forecast
            trend_forecast = sarimax_fitted_model.get_forecast(steps=horizon).predicted_mean
            
            # Fit exponential smoothing on seasonal component
            es_model = ExponentialSmoothing(seasonal, trend='add', seasonal_periods=7)
            es_fitted_model = es_model.fit()
            
            # Get forecast for seasonal component
            seasonal_forecast = es_fitted_model.forecast(steps=horizon)
            
            # Combine forecasts
            combined_forecast = trend_forecast + seasonal_forecast
            
            # Inverse transform and clip negative values
            inverse_transformed_forecast = pt.inverse_transform(combined_forecast.reshape(-1, 1)).flatten()
            forecasted_values[:, region] = np.maximum(inverse_transformed_forecast, 0)
        except Exception as e:
            try:
                # Fallback to simpler ARIMA model if SARIMAX fails
                arima_model = ARIMA(region_data, order=(1, 1, 1))
                arima_fitted_model = arima_model.fit()
                
                # Get forecast
                arima_forecast = arima_fitted_model.forecast(steps=horizon)
                
                # Inverse transform and clip negative values
                inverse_transformed_arima_forecast = pt.inverse_transform(arima_forecast.reshape(-1, 1)).flatten()
                forecasted_values[:, region] = np.maximum(inverse_transformed_arima_forecast, 0)
            except Exception as e:
                # Fallback to exponential smoothing if ARIMA also fails
                es_model = ExponentialSmoothing(region_data, trend='add', seasonal=None)
                es_fitted_model = es_model.fit()
                
                # Get forecast
                es_forecast = es_fitted_model.forecast(steps=horizon)
                
                # Clip negative values
                forecasted_values[:, region] = np.maximum(es_forecast, 0)
    
    return forecasted_values
```

### Explanation:
1. **Handling Missing Values**: The `SimpleImputer` is used to handle missing values by replacing them with the median of the respective time series.
2. **Power Transformation**: The `PowerTransformer` is applied to stabilize variance and make the data more normally distributed.
3. **Seasonal Decomposition**: The `seasonal_decompose` function is used to separate the time series into trend, seasonal, and residual components.
4. **SARIMAX Model**: A SARIMAX model is fitted to the trend and seasonal components separately.
5. **Exponential Smoothing**: Exponential smoothing is applied to the seasonal component to get a smoothed forecast.
6. **Combining Forecasts**: The forecasts from the trend and seasonal components are combined.
7. **Inverse Transform**: The transformed forecast is inversely transformed back to the original scale.
8. **Negative Value Handling**: Negative values are clipped to zero to ensure non-negativity.

This implementation aims to handle various scenarios such as missing values, non-stationarity, and seasonality while providing robust and reliable forecasts.
