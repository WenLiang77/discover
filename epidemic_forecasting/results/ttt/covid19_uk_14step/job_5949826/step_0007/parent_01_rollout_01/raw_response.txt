```python
import numpy as np
from statsmodels.tsa.holtwinters import ExponentialSmoothing
from statsmodels.tsa.arima.model import ARIMA
from statsmodels.tsa.statespace.sarimax import SARIMAX
from statsmodels.tsa.seasonal import seasonal_decompose
from sklearn.preprocessing import PowerTransformer
from sklearn.impute import SimpleImputer
from sklearn.metrics import mean_absolute_error, mean_squared_error, symmetric_mean_absolute_percentage_error

def covid_forecast(train_values, horizon, **kwargs):
    forecasted_values = np.zeros((horizon, train_values.shape[1]))
    imputer = SimpleImputer(strategy='mean')
    
    for region in range(train_values.shape[1]):
        region_data = train_values[:, region]
        
        # Impute missing values
        region_data = imputer.fit_transform(region_data.reshape(-1, 1)).flatten()
        
        try:
            # Decompose the data
            decomposition = seasonal_decompose(region_data, model='additive', period=7)
            trend = decomposition.trend
            seasonal = decomposition.seasonal
            
            # Fit SARIMAX model on the decomposed components
            sarimax_trend_model = SARIMAX(trend, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            sarimax_seasonal_model = SARIMAX(seasonal, order=(1, 1, 1), seasonal_order=(1, 1, 1, 7))
            
            sarimax_trend_fitted_model = sarimax_trend_model.fit(disp=False)
            sarimax_seasonal_fitted_model = sarimax_seasonal_model.fit(disp=False)
            
            # Get forecasts
            trend_forecast = sarimax_trend_fitted_model.get_forecast(steps=horizon).predicted_mean
            seasonal_forecast = sarimax_seasonal_fitted_model.get_forecast(steps=horizon).predicted_mean
            
            # Combine forecasts and apply inverse transform
            combined_forecast = trend_forecast + seasonal_forecast
            
            # Apply inverse transformation
            pt = PowerTransformer(method='yeo-johnson')
            pt.fit(region_data.reshape(-1, 1))
            inverse_transformed_forecast = pt.inverse_transform(combined_forecast.reshape(-1, 1)).flatten()
            
            # Ensure non-negative values
            forecasted_values[:, region] = np.maximum(inverse_transformed_forecast, 0)
        
        except Exception as e:
            try:
                # Fallback to ARIMA model
                arima_model = ARIMA(region_data, order=(1, 1, 1))
                arima_fitted_model = arima_model.fit(disp=False)
                forecast = arima_fitted_model.forecast(steps=horizon)
                
                # Apply inverse transformation
                inverse_transformed_forecast = pt.inverse_transform(forecast.reshape(-1, 1)).flatten()
                
                # Ensure non-negative values
                forecasted_values[:, region] = np.maximum(inverse_transformed_forecast, 0)
            
            except Exception as e:
                # Fallback to exponential smoothing
                es_model = ExponentialSmoothing(region_data, trend='add', seasonal=None)
                es_fitted_model = es_model.fit()
                forecast = es_fitted_model.forecast(steps=horizon)
                
                # Ensure non-negative values
                forecasted_values[:, region] = np.maximum(forecast, 0)
    
    return forecasted_values
```
